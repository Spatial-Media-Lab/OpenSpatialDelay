    ```ts
        await page.goto('https://spatialmedialab.org/about/', { waitUntil: 'domcontentloaded' });

      test('lists Andrew Rahman @external', async ({ page }) => {
        await expect(page.locator('body')).toContainText('Andrew Rahman');
      });

      test('lists Basel Naouri (joined 2025) @external', async ({ page }) => {
        await expect(page.locator('body')).toContainText('Basel Naouri');
      });

      test('lists Timo Bittner (co-founder, current president) @external', async ({ page }) => {
        await expect(page.locator('body')).toContainText('Timo Bittner');
      });

      test('no placeholder strings remain on the About page @external', async ({ page }) => {
        await expect(page.locator('body')).not.toContainText('TBD');
        await expect(page.locator('body')).not.toContainText('Coming soon');
        await expect(page.locator('body')).not.toContainText('Lorem ipsum');
      });
    });
    ```
