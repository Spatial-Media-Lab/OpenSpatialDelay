    ```toml
      to = "/index.html"

    ```

    The 404 fallback is not for SPAs (we're statically-exported) — it's a safety net so a typoed URL gets the `out/index.html` homepage rather than Netlify's default 404 chrome. `force = false` means real static files take precedence.

    **Step 2 — Create `&lt;SITE_ROOT&gt;/_headers`** with:
    ```
    /*
      Strict-Transport-Security: max-age=31536000; includeSubDomains
      X-Content-Type-Options: nosniff
      X-Frame-Options: DENY
      Referrer-Policy: strict-origin-when-cross-origin
      Permissions-Policy: camera=(), microphone=(), geolocation=()
      Content-Security-Policy: default-src 'self'; script-src 'self' https://tally.so https://widgets.tally.so; frame-src https://tally.so https://*.tally.so; style-src 'self' 'unsafe-inline'; img-src 'self' data: https:; connect-src 'self' https://tally.so; base-uri 'self'; form-action 'self' https://tally.so
    ```

    Notes on CSP:
    - `script-src` allows only our origin + Tally's CDN (required because `app/get-osd/page.tsx` injects `tally.so/widgets/embed.js`)
    - `frame-src` allows Tally to embed as an iframe
    - `style-src` allows `'unsafe-inline'` for Tailwind's dev-mode inline styles + Tally's iframe styling
    - NO `'unsafe-eval'` and NO broad `script-src *`

    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    # Production smoke verification for andrewrahman.com
    # Usage: ./scripts/verify-production.sh
    # Exits 0 if all checks pass; non-zero on first failure.

    ROOT="https://andrewrahman.com"
    FAILS=0

    check_status() {
      local path="$1"
      local expected="$2"
      local got
      got=$(curl -sI -o /dev/null -w "%{http_code}" "${ROOT}${path}")
      if [[ "$got" != "$expected" ]]; then
        echo "FAIL: ${path} returned ${got}, expected ${expected}" &gt;&amp;2
        FAILS=$((FAILS+1))
      else
        echo "PASS: ${path} -&gt; ${got}"
      fi
    }

    check_contains() {
      local path="$1"
      local needle="$2"
      if curl -s "${ROOT}${path}" | grep -q "$needle"; then
        echo "PASS: ${path} contains '${needle}'"
      else
        echo "FAIL: ${path} does NOT contain '${needle}'" &gt;&amp;2
        FAILS=$((FAILS+1))
      fi
    }

    check_header() {
      local header="$1"
      if curl -sI "${ROOT}/" | grep -qi "^${header}:"; then
        echo "PASS: header ${header} present"
      else
        echo "FAIL: header ${header} missing" &gt;&amp;2
        FAILS=$((FAILS+1))
      fi
    }

    echo "== HTTP status checks =="
    check_status "/" "200"
    check_status "/privacy/" "200"
    check_status "/get-osd/" "200"

    echo "== Content checks =="
    check_contains "/" "Andrew Rahman"
    check_contains "/" "Spatial Media Library"
    check_contains "/privacy/" "Privacy Policy"
    check_contains "/privacy/" "andrewjrahman@gmail.com"
    check_contains "/privacy/" "Tally Technologies SRL"
    check_contains "/get-osd/" "Get OpenSpatialDelay"

    echo "== Security header checks =="
    check_header "Strict-Transport-Security"
    check_header "X-Content-Type-Options"
    check_header "Content-Security-Policy"

    echo "== DNS check =="
    if dig andrewrahman.com +short A | grep -q "^75\\.2\\."; then
      echo "PASS: apex A resolves to 75.2.0.0/16 (Netlify)"
    else
      echo "FAIL: apex A does not resolve to Netlify IP range" &gt;&amp;2
      FAILS=$((FAILS+1))
    fi

    echo
    if [[ "$FAILS" -eq 0 ]]; then
      echo "ALL CHECKS PASSED"
      exit 0
    else
      echo "${FAILS} CHECK(S) FAILED"
      exit 1
    fi
    ```
