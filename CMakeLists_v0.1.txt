cmake_minimum_required(VERSION 3.22)
project(OpenSpatialDelay_v01 VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT DEFINED JUCE_DIR)
    set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../Spatial64/JUCE")
endif()

add_subdirectory(${JUCE_DIR} ${CMAKE_BINARY_DIR}/JUCE)

juce_add_plugin(OpenSpatialDelay_v01
    COMPANY_NAME "SpatialMediaLibrary"
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    PLUGIN_MANUFACTURER_CODE Smli
    PLUGIN_CODE Osd1
    FORMATS VST3 AU
    PRODUCT_NAME "OpenSpatialDelay v0.1"
    COPY_PLUGIN_AFTER_BUILD FALSE
)

juce_generate_juce_header(OpenSpatialDelay_v01)

target_sources(OpenSpatialDelay_v01
    PRIVATE
        Source/v0.1/PluginProcessor_v0.1.cpp
        Source/v0.1/PluginEditor_v0.1.cpp
)

# v0.1 files include "PluginProcessor.h" / "PluginEditor.h" (original names)
# Create wrapper headers so includes resolve correctly
file(WRITE "${CMAKE_BINARY_DIR}/v01_includes/PluginProcessor.h"
     "#include \"${CMAKE_CURRENT_SOURCE_DIR}/Source/v0.1/PluginProcessor_v0.1.h\"\n")
file(WRITE "${CMAKE_BINARY_DIR}/v01_includes/PluginEditor.h"
     "#include \"${CMAKE_CURRENT_SOURCE_DIR}/Source/v0.1/PluginEditor_v0.1.h\"\n")
target_include_directories(OpenSpatialDelay_v01 PRIVATE "${CMAKE_BINARY_DIR}/v01_includes")

target_compile_definitions(OpenSpatialDelay_v01
    PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(OpenSpatialDelay_v01
    PRIVATE
        juce::juce_audio_utils
        juce::juce_audio_plugin_client
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)

if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Build for Apple Silicon")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
endif()

if(APPLE)
    add_custom_command(TARGET OpenSpatialDelay_v01_VST3 POST_BUILD
        COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/scripts/install_plugins_v01.sh" "${CMAKE_BINARY_DIR}"
        COMMENT "Installing v0.1 AU and VST3 plugins..."
    )
endif()
